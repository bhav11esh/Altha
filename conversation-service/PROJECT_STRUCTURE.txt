# LOCAL DEVELOPMENT
docker-compose up -d              # Start services
docker-compose down               # Stop services
docker-compose logs -f app        # View logs
python test_client.py             # Run full demo

# DATABASE
psql $DATABASE_URL               # Connect to DB
psql -f init_db.sql              # Initialize schema

# TESTING
pytest test_app.py -v            # Run unit tests
bash test_conversation_service.sh # Run integration tests

# API ACCESS
curl http://localhost:8000/health                    # Health check
curl http://localhost:8000/docs                      # Swagger UI
open http://localhost:8000/redoc                     # ReDoc

# ENVIRONMENT
cp .env.example .env.local       # Create local env
source .env.local                # Load env vars

# PRODUCTION
docker build -t conversation-service .
docker run -p 8000:8000 conversation-service
docker-compose -f docker-compose.prod.yml up -d
